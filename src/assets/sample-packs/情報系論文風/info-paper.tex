\newcommand{\makepapertitle}{%
  \twocolumn[{
    \begin{center}
      {\LARGE\bfseries \paperTitle\par}

      \fi

      \ifx\paperAffiliation\@empty\else
        \vspace{0.3em}{\small \paperAffiliation\par}
      \fi
    \end{center}
    \vspace{0.8em}
    \noindent\textbf{概要}\quad \paperAbstract\par
    \vspace{0.5em}
    \noindent\textbf{キーワード}\quad \paperKeywords\par
    \vspace{1.2em}
  }]
}
