% preamble.tex — 縦書き二段組向けプリアンブル（小説組版・ADR-007 defaults 方式サンプル）
%
% builtin モードの DEFAULT_LATEX_PREAMBLE を縦書き ltjtarticle + twocolumn（小説用途）
% 向けに最適化した版。defaults 方式では headerIncludes がスキップされるため、
% MdTex 固有 Lua フィルタ（callout/mermaid/docx）が前提とする
% \passthrough・listings 等を含め、defaults 方式で必須となるパッケージを自前ロードする。
% さらに小説組版（ルビ・圏点・話者名字取り）に必要なマクロを整える。
% 詳細は README.md「学んだ落とし穴」を参照。

% --- Pandoc raw_attribute 互換（必須） ---
% Pandoc は Markdown 内の {=latex} 等の raw block や生 LaTeX を \passthrough{...}
% で包んで出力する（MdTex は入力形式に +raw_attribute を常時指定）。
% builtin モードでは DEFAULT_LATEX_PREAMBLE がこの定義を担うが、defaults 方式では
% headerIncludes がスキップされるため、ここで必ず定義すること。
\providecommand{\passthrough}[1]{#1}

% --- フォント ---
% 和文フォント: ltjtarticle 既定を尊重（Noto 系に寄せるなら以下を有効化）
% \setmainjfont{Noto Serif CJK JP}
% \setsansjfont{Noto Sans CJK JP}

% 欧文（数字含む）は和文に自然に混ざるよう大文字基準でスケール。
% 縦組では自動的に回転して配置される。
\usepackage{luatexja-fontspec}
\setmainfont[Scale=MatchUppercase]{Noto Serif}
\setsansfont[Scale=MatchUppercase]{Noto Sans}

% --- 数式 ---
\usepackage{amsmath,amssymb}

% --- コード（listings）---
% callout/mermaid/docx の MdTex 固有 Lua フィルタは、キャプション内の
% インラインコード等を \lstinline!...! で出力する。このコマンドは listings パッケージ
% が定義するため、defaults 方式でも必ずロードすること（builtin では DEFAULT_LATEX_PREAMBLE 経由）。
\usepackage{listings}
\lstset{
  basicstyle=\ttfamily,
  columns=fullflexible,
  keepspaces=true,
  breaklines=true
}

% --- 小説組版（ルビ・圏点・縦書き補助）---
% \ruby / \kenten は luatexja-ruby が提供。縦書き ltjtarticle で必須。
\usepackage{luatexja-ruby}

% --- Markdown 見出しを小説向けに控えめに整える ---
% サンプル本文は通常の Markdown として # / ## / ###### を使う。
% Pandoc が出力する \section / \subsection / \paragraph をここで組版調整する。
\usepackage{titlesec}
\titleformat{\section}{\normalsize\bfseries}{ }{0pt}{}
\titlespacing*{\section}{0pt}{0pt}{6mm}
\titleformat{\subsection}{\normalsize\bfseries}{ }{0pt}{}
\titlespacing*{\subsection}{0pt}{3mm}{2mm}
\titleformat{\paragraph}{\scriptsize}{ }{0pt}{}
\titlespacing*{\paragraph}{0pt}{0pt}{3mm}
\titleformat{\subparagraph}{\scriptsize}{ }{0pt}{}
\titlespacing*{\subparagraph}{0pt}{0pt}{6mm}

% --- 図表 ---
\usepackage{graphicx}
\usepackage{booktabs}
\usepackage{array}

% --- 参照 ---
\usepackage[unicode,hypertexnames=false]{hyperref}

% --- 本文サイズ・行送り ---
% ltjtarticle は 9pt オプションを無視するため、本文開始時に小説向けサイズを明示する。
% 8.8pt / 12pt は A5 二段組でも詰まりすぎず、既定よりかなり小説らしい密度になる。
\AtBeginDocument{\fontsize{8.8pt}{12pt}\selectfont\setlength{\parindent}{1em}}
\setlength{\parskip}{0pt}
\linespread{0.92}

% --- 二段組の段間調整 ---
% 余白を詰める一方、段間は読み分けられる程度に確保する。
% NOTE: LuaTeX-ja の和文単位 zw は begin{document} 後に定義されるため、
% preamble では使えない。pt / mm 等の絶対単位で指定する。
\setlength{\columnsep}{6mm}

% ===========================================================================
% 章扉・表題ページ（装飾と文字の分離 / 商業組版ワークフロー）
% ---------------------------------------------------------------------------
% 原則: 「装飾」は画像/PDF で作り、「文字」は LaTeX で組む。最終 PDF で合成する。
%   装飾レイヤ = \chapterbg（既定は TikZ の最小装飾。実運用では文字なし PDF に差替）
%   文字レイヤ   = 第N章 / 章題を TikZ ノードで配置（ベクター・字面管理）
%
% 【縦組 ltjtarticle + twocolumn での設計上の注意（重要）】
%   * メインテキストフロー内の tikzpicture は縦組の方向設定を引き継ぐため、
%     TikZ 座標軸がページに対し90°回転する（tikz +x=ページ下、tikz +y=ページ右）。
%     さらに \begin{center} による水平中央寄せが位置依存で安定しない。
%   * 一方、背景描画用の eso-pic \put(0,0){...} の中の tikzpicture は
%     「絶対ページ座標（回転なし）」で動く。(0,0)=左下、+x=右、+y=上。
%   * よって本パックでは、章扉・表題の「文字」も背景と同じ絶対座標の
%     tikzpicture に載せる。複数文字ノードは [align=center] + "\\" で縦に積む
%     （Lua フィルタが ::: novel-chapter の章題を1字ずつ "\\" 区切りで渡す）。
%     これで 1パス確定・縦組/段組に依存しない正確な配置になる。
% ===========================================================================
\usepackage{tikz}
\usetikzlibrary{calc}
\usepackage{eso-pic}

% --- 装飾背景（章扉・表題ページ共通のアンビエント装飾。文字は含めない）---
% 既定: TikZ で紙面いっぱいの薄墨・淡円を描く（パック単独でコンパイルを通すため）。
% 【画像に差し替える場合】preamble を直接編集せず、defaults.yaml の *metadata:* で指定:
%     metadata:
%       chapter-bg-image: chapter-bg.pdf
% chapter-bg.lua が実ファイルを探索して絶対パスに解決し、テンプレート
% （tate-twocolumn.tex）が \renewcommand{\chapterbg}{\includegraphics{...}} を生成する。
% 画像はパックフォルダ（resource-path: ${.}）に置けば発見する。
\newcommand{\chapterbg}{%
  \begin{tikzpicture}
    \useasboundingbox (0,0) rectangle (\paperwidth,\paperheight);
    \fill[black!4] (0,0) rectangle (\paperwidth,\paperheight);% 薄い地色
    \fill[black!7] (\paperwidth/2,\paperheight/2) circle (34mm);% 中央の淡円（薄墨のにじみ）
  \end{tikzpicture}%
}

% 装飾背景を「このページだけ」に敷く（\AddToShipoutPictureBG* の * は現ページ限定）。
\newcommand{\chapterbgshipout}{%
  \AddToShipoutPictureBG*{\put(0,0){\chapterbg}}%
}

% 第N章 の N（漢数字1〜10、超過は算用数字）。
\newcounter{nchapter}
\newcommand{\nchapterkanji}{%
  \ifcase\value{nchapter}\or
    一\or 二\or 三\or 四\or 五\or
    六\or 七\or 八\or 九\or 十\else \arabic{nchapter}%
  \fi
}
\newcommand{\nchaptertwodigits}{\ifnum\value{nchapter}<10 0\fi\arabic{nchapter}}

% 横組の小さなテキストブロックを縦組文書中の TikZ ノードに置くための補助。
\newcommand{\chapterYokoBox}[2]{%
  \begin{minipage}{#1}\yoko #2\end{minipage}%
}

% --- 表題ページ（参照画像風のモダン・エディトリアル配置）---
% aozora-ruby.lua の ::: novel-title が \noveltitle[副題]{題} に変換する。
% 背景は画像/PDF、文字は LaTeX で別レイヤーに配置する。
\newcommand{\noveltitle}[6]{%
  \clearpage
  \thispagestyle{empty}%
  \chapterbgshipout
  \AddToShipoutPictureBG*{%
    \put(0,0){%
      \begin{tikzpicture}
        \useasboundingbox (0,0) rectangle (\paperwidth,\paperheight);
        % 左上ラベル（Markdown の novel-title 第3段落）
        \draw[line width=0.35pt] (10mm,\paperheight-10mm) -- (22mm,\paperheight-10mm);
        \node[anchor=north west] at (10mm,\paperheight-13mm)
          {\chapterYokoBox{28mm}{\sffamily\bfseries\fontsize{6pt}{8pt}\selectfont #3}};
        % 上部キャプション（Markdown 第5段落）
        \draw[line width=0.25pt] (62mm,\paperheight-8mm) -- (62mm,\paperheight-20mm);
        \node[anchor=north west] at (70mm,\paperheight-9mm)
          {\chapterYokoBox{48mm}{\sffamily\bfseries\fontsize{5.5pt}{7pt}\selectfont #5}};
        % 大きな装飾番号（Markdown 第4段落）
        \node[anchor=north west,inner sep=0pt] at (9mm,\paperheight-42mm)
          {\chapterYokoBox{44mm}{\fontsize{64pt}{64pt}\selectfont #4}};
        \draw[line width=0.25pt,black!65] (54mm,\paperheight-78mm) -- (88mm,\paperheight-44mm);
        % 表題・副題（Markdown 第1・第2段落）
        \node[anchor=north west] at (10mm,\paperheight-92mm)
          {\chapterYokoBox{72mm}{\gtfamily\bfseries\fontsize{18pt}{26pt}\selectfont #1}};
        \draw[line width=0.3pt] (10mm,\paperheight-118mm) -- (82mm,\paperheight-118mm);
        \node[anchor=north west] at (10mm,\paperheight-124mm)
          {\chapterYokoBox{72mm}{\rmfamily\bfseries\fontsize{8pt}{11pt}\selectfont #2}};
        % 下部キャプション（Markdown 第6段落）
        \node[anchor=south west] at (10mm,13mm)
          {\chapterYokoBox{70mm}{\sffamily\bfseries\fontsize{5.5pt}{7pt}\selectfont #6}};
      \end{tikzpicture}%
    }%
  }%
  \mbox{}\clearpage
}

% --- 章扉ページ（参照画像風：大きな章番号＋横組章題）---

  \AddToShipoutPictureBG*{%

      \begin{tikzpicture}
        \useasboundingbox (0,0) rectangle (\paperwidth,\paperheight);
        % 左上ラベル（Markdown 第5段落）

        \node[anchor=north west] at (10mm,\paperheight-13mm)
          {\chapterYokoBox{28mm}{\sffamily\bfseries\fontsize{6pt}{8pt}\selectfont #5}};
        % 上部キャプション・右上番号（Markdown 第6・第7段落）
        \draw[line width=0.25pt] (62mm,\paperheight-8mm) -- (62mm,\paperheight-20mm);
        \node[anchor=north west] at (70mm,\paperheight-9mm)

          {\chapterYokoBox{76mm}{\gtfamily\bfseries\fontsize{18pt}{28pt}\selectfont #1}};

      \end{tikzpicture}%

}
